## arm command policy (node bin/fm-arm-command-policy.mjs --root . --home <tmp> --command ...)
bash -s sentinel <<< 'bin/fm-watch.sh'             -> deny	watcher-nested	a protected watcher command must not run through a wrapper, substitution, or compound command
sh -s sentinel <<< 'bin/fm-watch.sh'               -> deny	watcher-nested	a protected watcher command must not run through a wrapper, substitution, or compound command
bash -s -- -c harmless <<< 'bin/fm-watch.sh'       -> deny	watcher-nested	a protected watcher command must not run through a wrapper, substitution, or compound command
bash -s bin/fm-watch.sh                            -> deny	watcher-nested	a protected watcher command must not run through a wrapper, substitution, or compound command
bash -s sentinel <<< 'echo fm-watch.sh'            -> allow
bash -- -s sentinel <<< 'bin/fm-watch.sh'          -> allow
