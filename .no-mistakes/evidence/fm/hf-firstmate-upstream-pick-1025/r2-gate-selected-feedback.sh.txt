    def task(self, task, extra='', mode='direct-PR', target=True):
        body = f'kind=ship\nmode={mode}\npr={URL}\nspawn_gen=one\n'
        if target:
            body += f'window=sess:fm-{task}\nharness=claude\n'
        (self.state / f'{task}.meta').write_text(body + extra)

    def save(self):
        self.payload.write_text(json.dumps(self.data))

    def call(self, *args, ok=True):
        p = subprocess.run([CLI, *args], env=self.env, text=True, capture_output=True, timeout=60)
        if ok:
            self.assertEqual(p.returncode, 0, p.stderr + p.stdout)
        else:
            self.assertNotEqual(p.returncode, 0)
        return p.stdout

    def poll(self):
        self.save()
        return self.call('poll')

    def messages(self, task='t1', handled=False):
        base = self.state / f'{task}.inbox'
        if handled:
            base /= 'handled'
        return [p.read_text() for p in sorted(base.glob('*.msg'))]

    def pipeline_task(self, pr=URL, outcome=''):
        wt = self.home / 'work'
        subprocess.run(['git', 'init', '-q', str(wt)], check=True, env=self.env)
        subprocess.run(['git', '-C', str(wt), '-c', 'user.name=Fixture', '-c', 'user.email=f@example.test',
                        'commit', '-q', '--allow-empty', '-m', 'fixture'], check=True, env=self.env)
        subprocess.run(['git', '-C', str(wt), 'checkout', '-qb', 'fm/t1'], check=True, env=self.env)
        head = subprocess.check_output(['git', '-C', str(wt), 'rev-parse', 'HEAD'], text=True).strip()
        self.task('t1', f'pr={pr}\nworktree={wt}\nbranch=fm/t1\n', mode='no-mistakes')
        status = 'completed' if outcome else 'running'
        self.overview.write_text(f'count: 1 of 1 total\nruns[1]{{id,branch,status,head,pr}}:\n'
                                 f'  run1,fm/t1,{status},{head},{pr}\n')
        text = f'run:\n  id: "run1"\n  branch: fm/t1\n  status: {status}\n  head_sha: {head}\n  pr: "{pr}"\n'
        if outcome:
            text += f'outcome: {outcome}\n'
        self.nm.write_text(text)
        self.data['repos/o/r/pulls/7']['head']['sha'] = head
        self.data[f'repos/o/r/commits/{head}/check-runs'] = {'check_runs': []}
        self.data[f'repos/o/r/commits/{head}/status'] = {'statuses': []}
        return text, head

    def test_feedback_bots_pagination_self_filter_and_handled_replay(self):
        self.data['repos/o/r/pulls/7/reviews'] = [self.row(1, state='APPROVED'),
                                                self.row(2, 'WORKER', state='COMMENTED')]
        self.data['repos/o/r/pulls/7/comments'] = [self.row(3, 'reviewbot[bot]')]
        self.data['repos/o/r/issues/7/comments'] = [self.row(n) for n in range(10, 22)]
        out = self.poll()
        self.assertEqual(out.count('\n'), 1)
        self.assertIn('tasks=t1 relaunch=none errors=none', out)
        self.assertEqual(len(self.messages()), 1)
        joined = '\n'.join(self.messages())
        self.assertEqual(joined.count('GitHub PR feedback event:'), 14)
        self.assertIn('#pullrequestreview-1', joined)
        self.assertIn('#discussion_r3', joined)
        self.assertIn('#issuecomment-21', joined)
        self.assertNotIn('#pullrequestreview-2', joined)
        self.assertNotIn('UNTRUSTED', joined)
        rings = self.log.read_text()
        self.assertEqual(len(rings.splitlines()), 1)
        self.assertNotIn('github.com', rings)
        self.assertIn('Firstmate instruction waiting', rings)
        for p in (self.state / 't1.inbox').glob('*.msg'):
            p.rename(p.parent / 'handled' / p.name)
        self.assertEqual(self.poll(), '')
        self.assertEqual(self.log.read_text(), rings)
        self.data['repos/o/r/issues/7/comments'][0]['updated_at'] = '2026-10-25T01:00:00Z'
        self.assertIn('tasks=t1', self.poll())
        self.assertEqual(len(self.messages()), 1)
        self.assertEqual(len(self.messages(handled=True)), 1)

    def test_ci_results_ignore_pending_churn_and_order_but_include_reruns(self):
        key = f'repos/o/r/commits/{HEAD}/check-runs'
        self.data[key]['check_runs'] = [self.check('queued', None)]
        self.assertEqual(self.poll(), '')
        self.data[key]['check_runs'] = [self.check('in_progress', None)]

        self.assertIn('tasks=t1', self.poll())
        self.assertIn('state=red', self.messages()[0])

        self.assertIn('tasks=t1', self.poll())
        self.assertIn('state=green', self.messages()[1])

        self.assertIn('tasks=t1', self.poll())
        self.assertEqual(len(self.messages()), 3)

    def test_registration_binding_disarm_and_unsafe_inbox(self):
        self.call('enabled')
        shim = self.state / 'pr-feedback.check.sh'
        shim.write_text(shim.read_text() + '# tampered\n')
        self.call('enabled', ok=False)
        self.call('poll', ok=False)
        self.call('arm')
        outside = self.home / 'outside'
        outside.mkdir()
        (self.state / 't1.inbox').symlink_to(outside, target_is_directory=True)
        digest = hashlib.sha256((self.state / 't1.meta').read_bytes()).hexdigest()
        self.call('deliver', 't1', digest, 'pointer', ok=False)
        self.data['repos/o/r/issues/7/comments'] = [self.row(1)]
        self.assertIn('errors=t1', self.poll())
        self.assertEqual(list(outside.iterdir()), [])
        self.call('disarm')
        self.call('enabled', ok=False)
        self.assertFalse(shim.exists())
        self.assertFalse((self.state / 'pr-feedback.check-trust').exists())

    def test_interrupted_sweep_surfaces_unfinished_tasks(self):
        self.env['FM_TEST_API_DELAY'] = '1'
        process = subprocess.Popen([CLI, 'poll'], env=self.env, text=True,
                                   stdout=subprocess.PIPE, stderr=subprocess.PIPE,
                                   start_new_session=True)
        try:
            for _ in range(500):
                if self.api_log.exists():
                    break
                time.sleep(0.02)
            self.assertTrue(self.api_log.exists(), 'the observer never reached its API read')
            os.killpg(process.pid, signal.SIGTERM)
            out, err = process.communicate(timeout=10)
            self.assertIn('interrupted=t1', out, err)
            self.assertIn('increase FM_CHECK_TIMEOUT', out)
            self.assertEqual(out.count('pr-feedback batch:'), 1)
        finally:
            if process.poll() is None:
                os.killpg(process.pid, signal.SIGKILL)
                process.communicate(timeout=10)

    def test_scalar_and_mapping_gate_run_findings_rounds_and_handled_replay(self):
        text, _ = self.pipeline_task(pr='')
        round_one = '  findings[1]{id,severity,file,line,action,description}:\n    R1,error,a.py,1,ask-user,first\n'
        round_two = round_one.replace('R1,', 'R2,')
        shapes = [('gate: review\n', 'awaiting_approval'),
                  ('gate: review\n', 'fix_review'),
                  ('gate: review\n', 'running'),
                  ('gate:\n  step: review\n  status: fix_review\n', 'running')]
        for gate, status in shapes:
            with self.subTest(gate=gate, status=status):
                current = text.replace('status: running', f'status: {status}')
                self.nm.write_text(current + round_one + gate)
                before = len(self.messages(handled=True))
                self.assertIn('tasks=t1', self.poll())
                self.assertIn('run=run1 step=review', self.messages()[0])
                for p in (self.state / 't1.inbox').glob('*.msg'):
                    p.rename(p.parent / 'handled' / p.name)
                self.nm.write_text(current.replace('  pr:', '  awaiting_agent: parked 20m\n  pr:') + round_one + gate)
                self.assertEqual(self.poll(), '')
                self.nm.write_text(current + round_two + gate)
                self.assertIn('tasks=t1', self.poll())
                self.assertEqual(len(self.messages(handled=True)), before + 1)
                self.assertEqual(len(self.messages()), 1)
                self.assertEqual(self.poll(), '')
                self.nm.write_text(current + '  round: 2\n' + round_two + gate)
                self.assertIn('tasks=t1', self.poll())
                self.assertEqual(len(self.messages()), 2)
                self.assertEqual(self.poll(), '')
                for p in (self.state / 't1.inbox').rglob('*.msg'):
                    p.unlink()

    def test_identical_gate_findings_at_new_attributed_heads_notify_once(self):
        text, head = self.pipeline_task(pr='')
        next_head = 'b' * 40
        absent = subprocess.run(['git', '-C', str(self.home / 'work'), 'cat-file', '-e', next_head],
                                env=self.env, capture_output=True)
        self.assertNotEqual(absent.returncode, 0)
        findings = '  findings[1]{id,severity,action}:\n    R1,error,auto-fix\n'
        custody = 'branch_sync:\n  state: pipeline_owned\n'
        for field, changed in (('head_sha', next_head), ('head', next_head[:12])):
            for gate in ('gate: review\n', 'gate:\n  step: review\n  status: fix_review\n'):
                with self.subTest(field=field, gate=gate):
                    current = (text.replace('head_sha:', f'{field}:')
                               .replace('status: running', 'status: fix_review') + findings + custody + gate)
                    self.nm.write_text(current)
                    self.assertIn('tasks=t1 relaunch=none errors=none', self.poll())
                    for record in (self.state / 't1.inbox').glob('*.msg'):
                        record.rename(record.parent / 'handled' / record.name)
                    self.assertEqual(self.poll(), '')
                    self.nm.write_text(current.replace(head, changed))
                    self.assertIn('tasks=t1 relaunch=none errors=none', self.poll())
                    self.assertEqual(len(self.messages()), 1)
                    self.assertEqual(len(self.messages(handled=True)), 1)

    def test_terminal_receipts_do_not_bind_unseen_runs_or_reused_work(self):
        text, local_head = self.pipeline_task(pr='https://review.example/c/o/r/+/7')
        next_head = 'b' * 40
        current = text.replace(local_head, next_head)
        terminal = current.replace('status: running', 'status: completed') + 'outcome: passed-with-skips\n'
        completed = ('count: 1 of 1 total\nruns[1]{id,branch,status,head,pr}:\n'
                     f'  run1,fm/t1,completed,{next_head},\n')
        self.overview.write_text(completed)
        self.nm.write_text(terminal)
        self.assertEqual(self.poll(), '')
        self.overview.write_text(completed.replace('completed,', 'running,'))
        self.nm.write_text(current + 'branch_sync:\n  state: pipeline_owned\ngate: review\n')
        self.assertIn('tasks=t1', self.poll())
        self.nm.write_text(terminal)
        self.overview.write_text(completed.replace('run1,', 'run2,'))
        (self.home / 'other.toon').write_text(terminal.replace('run1', 'run2'))
        self.assertEqual(self.poll(), '')
        self.overview.write_text(completed.replace('fm/t1', 'fm/reused'))
        self.nm.write_text(terminal.replace('fm/t1', 'fm/reused'))
        meta = self.state / 't1.meta'
        original = meta.read_text()

                       check=True, env=self.env)
        self.assertEqual(self.poll(), '')
        self.assertEqual(len(self.messages()), 1)
        self.assertNotIn('No-mistakes outcome event:', self.messages()[0])
        self.assertFalse(self.api_log.exists())

    def test_no_checks_markers_never_emit_green_feedback(self):
        # Accepted PR contract: an absent check is unverified, even when a
        # previous marker was green or the no_ci sentence says checks passed.
        text, head = self.pipeline_task()
        self.nm.write_text(text + 'steps[1]{step,status,findings,duration_ms}:\n  ci,running,0,0\n')
        green = 'all CI checks passed - still monitoring until merged or closed\n'
        for marker in ('no CI checks reported - still monitoring until merged or closed\n',
                       'repository declares no CI (no_ci: true) - treating as all checks passed - '
                       'still monitoring until merged or closed\n'):
            with self.subTest(marker=marker):
                self.ci_log.write_text(green + marker)
                self.assertEqual(self.poll(), '')
                self.assertEqual(self.messages(), [])
        self.ci_log.write_text(marker + green)
        self.assertIn('tasks=t1', self.poll())
        self.assertIn(f'GitHub PR CI event: {URL} head={head} state=green', self.messages()[0])

    def test_no_checks_readiness_requires_current_attributed_pipeline_evidence(self):
        text, head = self.pipeline_task(pr='')
        monitoring = text.replace('  pr: ""', f'  pr: "{URL}"') + (
            'steps[1]{step,status,findings,duration_ms}:\n  ci,running,0,0\n')

        ready = 'all CI checks passed - still monitoring until merged or closed\n'
        pending = 'no CI checks reported yet, waiting for checks to register...\n'
        self.assertEqual(self.poll(), '')
        self.ci_log.unlink()
        self.assertEqual(self.poll(), '')
        self.ci_log.write_text(pending)
        self.assertEqual(self.poll(), '')
        self.ci_log.write_text(ready + pending)
        self.assertEqual(self.poll(), '')
        self.ci_log.write_text(pending + ready + 'base branch advanced, re-arming CI monitor timeout\n')
        self.nm.write_text(monitoring.replace(head, 'b' * 40))
        self.assertEqual(self.poll(), '')
        self.nm.write_text(monitoring)
        self.task('t1', f'worktree={self.home / "work"}\nbranch=fm/t1\n', mode='no-mistakes')
        self.nm.write_text(monitoring.replace(URL, 'https://github.com/o/r/pull/8'))

        self.assertIn(f'GitHub PR CI event: {URL} head={head} state=green', self.messages()[0])

        self.ci_log.write_text(ready + 'issues detected, fixing\n')
        self.assertEqual(self.poll(), '')

    def test_pipeline_no_checks_evidence_never_overrides_pending_or_failed_checks(self):
        text, head = self.pipeline_task()
        self.nm.write_text(text.replace('status: running', 'status: ci'))
        self.ci_log.write_text('all CI checks passed - still monitoring until merged or closed\n')
        self.data[f'repos/o/r/commits/{head}/check-runs']['check_runs'] = [self.check('queued', None)]
        self.assertEqual(self.poll(), '')
        self.data[f'repos/o/r/commits/{head}/check-runs']['check_runs'] = [self.check(conclusion='failure')]
        self.assertIn('tasks=t1', self.poll())
        self.assertIn('state=red', self.messages()[0])
        self.assertNotIn('state=green', self.messages()[0])
        self.data[f'repos/o/r/commits/{head}/check-runs']['check_runs'] = []
        self.data[f'repos/o/r/commits/{head}/status']['statuses'] = [{'id': 9, 'state': 'pending'}]
        self.assertEqual(self.poll(), '')
        self.data[f'repos/o/r/commits/{head}/status']['statuses'] = [{'id': 10, 'state': 'failure'}]
        self.assertIn('tasks=t1', self.poll())
        self.assertNotIn('state=green', '\n'.join(self.messages()))
        self.data[f'repos/o/r/commits/{head}/status']['statuses'] = []
        self.assertIn('tasks=t1', self.poll())
        self.assertIn('state=green', '\n'.join(self.messages()))
